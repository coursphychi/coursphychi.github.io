Ratio d'aspect ---------- */ -}} {{- $ratio := "16x9" -}} {{- if .IsNamedParams -}}{{- $ratio = .Get "ratio" | default "16x9" -}}{{- end -}} {{- $ratioCSS := "16 / 9" -}} {{- if eq $ratio "4x3" -}} {{- $ratioCSS = "4 / 3" -}} {{- else if in (slice "9x16" "short" "portrait" "vertical") $ratio -}} {{- $ratioCSS = "9 / 16" -}} {{- else if ne $ratio "16x9" -}} {{- warnf "youtube-plus : ratio %q inconnu, repli sur 16x9 (%s)" $ratio .Position -}} {{- end -}} {{- /* ---------- 3. Construction de l'URL ---------- */ -}} {{- $params := slice -}} {{- with $times.Get "start" -}}{{- $params = $params | append (printf "start=%s" .) -}}{{- end -}} {{- with $times.Get "end" -}}{{- $params = $params | append (printf "end=%s" .) -}}{{- end -}} {{- $src := printf "https://www.youtube-nocookie.com/embed/%s" $id -}} {{- if $params -}}{{- $src = printf "%s?%s" $src (delimit $params "&") -}}{{- end -}} {{- /* ---------- 7. Rendu ---------- */ -}} {{- /* ---------- 7.
